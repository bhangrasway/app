    <script>
        // ===== SUPABASE CONFIGURATION (same project as the admin dashboard) =====
        const SUPABASE_URL = 'https://geeqcyfnkgamcjclulay.supabase.co';
        const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6ImdlZXFjeWZua2dhbWNqY2x1bGF5Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODIwNDYyNjMsImV4cCI6MjA5NzYyMjI2M30.FPEQGtuDMxWK8GnPNQfh88ubXlQ2EIkV7X6FZN_TQpo';
        const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        // Bhangra Sway, Cremona (from the studio's Google Maps listing)
        const STUDIO_LAT = 45.1395108;
        const STUDIO_LNG = 10.0288616;
        const STUDIO_RADIUS_METERS = 500;

        // ===== Identification =====
        // No member names ever reach this page up-front: the member types their
        // phone number and the find_members_by_phone RPC returns only the
        // member(s) registered under that exact number (their own household).
        // The phone number is then re-verified inside the database by every
        // action RPC, so the id alone is never trusted.
        let mode = 'checkin';            // 'checkin' | 'notify'
        let notifyReason = '';
        let enteredPhone = '';
        let selectedMember = null;
        let awayFromHome = false;
        // Last-shown dynamic texts, kept as {key, vars} so a language switch
        // can re-render whatever screen is currently visible.
        let branchKey = 'checking_in_pending';
        let closedMsg = null;
        let locBlockStatus = 'too-far'; // which variant the location screen shows

        function show(id) {
            ['loadingState', 'closedState', 'tooFarState', 'phoneState', 'pickState', 'confirmState', 'resultState'].forEach((s) => {
                document.getElementById(s).classList.toggle('hidden', s !== id);
            });

            // Give the browser's Back button something to land on: the first time
            // we leave the phone screen (the "home" screen of this page), push one
            // history entry. Popping it (Back) re-runs init(), which re-checks
            // everything fresh and lands back on whichever screen is correct now.
            if (id === 'phoneState' || id === 'loadingState') {
                awayFromHome = false;
            } else if (!awayFromHome) {
                awayFromHome = true;
                history.pushState({ checkin: true }, '', location.href);
            }
        }

        function formatTime(t) {
            const [h, m] = String(t).split(':').map(Number);
            const period = h >= 12 ? 'PM' : 'AM';
            const hour12 = ((h + 11) % 12) + 1;
            return `${hour12}:${String(m).padStart(2, '0')} ${period}`;
        }

        // Surnames are shown as initials only, the confirm screen can be read
        // over a shoulder at the kiosk, and siblings in the household picker
        // don't need each other's full surnames on screen either.
        function maskName(full) {
            const parts = String(full || '').trim().split(/\s+/);
            if (parts.length <= 1) return parts[0] || '';
            return parts[0] + ' ' + parts.slice(1).map((p) => p[0].toUpperCase() + '.').join(' ');
        }

        function escapeHtml(str) {
            const div = document.createElement('div');
            div.textContent = str;
            return div.innerHTML;
        }

        // ===== Location is requested automatically on every visit =====
        // Self check-in requires a CONFIRMED location at the studio. A device
        // that won't share its location (denied / unsupported / timed out) is
        // routed to the request flow, where the instructor — who can see who's
        // actually in the room — approves with one tap. The old soft-pass let
        // anyone check in from home by simply denying the location prompt.
        function haversineMeters(lat1, lng1, lat2, lng2) {
            const R = 6371000;
            const toRad = (d) => (d * Math.PI) / 180;
            const dLat = toRad(lat2 - lat1);
            const dLng = toRad(lng2 - lng1);
            const a = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
            return 2 * R * Math.asin(Math.sqrt(a));
        }

        // Returns 'confirmed' (at the studio), 'too-far' (definitely not), or
        // 'unknown' (denied / unsupported / timed out, treated as a soft pass).
        async function checkLocationStatus() {
            if (!navigator.geolocation) return 'unknown';

            return await new Promise((resolve) => {
                navigator.geolocation.getCurrentPosition(
                    (pos) => {
                        const distance = haversineMeters(
                            pos.coords.latitude, pos.coords.longitude, STUDIO_LAT, STUDIO_LNG
                        );
                        resolve(distance <= STUDIO_RADIUS_METERS ? 'confirmed' : 'too-far');
                    },
                    () => resolve('unknown'),
                    { timeout: 8000, maximumAge: 60000 }
                );
            });
        }

        async function confirmBranchSilently() {
            const status = await checkLocationStatus();
            const branchLine = document.getElementById('branchLine');
            branchKey = status === 'confirmed' ? 'branch_confirmed' : 'branch_default';
            branchLine.textContent = t(branchKey);
            branchLine.classList.remove('pending');
            return status;
        }

        // ===== Main flow =====
        async function init() {
            const { data: settings, error: settingsError } = await supabaseClient
                .from('checkin_settings')
                .select('*')
                .eq('id', 1)
                .single();

            if (settingsError || !settings) {
                closedMsg = { key: 'not_setup', vars: null };
                document.getElementById('closedMessage').textContent = t('not_setup');
                show('closedState');
                return;
            }

            const now = new Date();
            const day = now.getDay();
            const hhmmss = now.toTimeString().slice(0, 8);
            const activeDays = settings.active_days || [];
            // Window may cross midnight (e.g. 22:00 → 01:00): then it's open when
            // the time is AFTER start OR BEFORE end, instead of between the two.
            const inWindow = settings.start_time <= settings.end_time
                ? (hhmmss >= settings.start_time && hhmmss <= settings.end_time)
                : (hhmmss >= settings.start_time || hhmmss <= settings.end_time);
            const isOpen = activeDays.includes(day) && inWindow;

            if (!isOpen) {
                closedMsg = { key: 'closed_msg', days: activeDays, start: settings.start_time, end: settings.end_time };
                renderClosedMessage();
                show('closedState');
                return;
            }

            const locationStatus = await confirmBranchSilently();
            if (locationStatus !== 'confirmed') {
                locBlockStatus = locationStatus;
                renderLocBlock();
                show('tooFarState');
                return;
            }

            startIdentify('checkin', '');
        }

        // The location screen covers two cases with different wording: the
        // phone POSITIVELY reported being >500m away, or it simply didn't
        // share a usable location (denied / unsupported / timed out).
        function renderLocBlock() {
            const unknown = locBlockStatus === 'unknown';
            document.getElementById('locBlockHeading').textContent = t(unknown ? 'loc_unknown_heading' : 'too_far_heading');
            document.getElementById('locBlockBody').textContent = t(unknown ? 'loc_unknown_body' : 'too_far_body');
            document.getElementById('locRetryBtn').textContent = t('retry_location');
        }

        function renderClosedMessage() {
            if (!closedMsg) return;
            if (closedMsg.key === 'not_setup') {
                document.getElementById('closedMessage').textContent = t('not_setup');
                return;
            }
            const dayList = closedMsg.days.map((d) => tDay(d)).join(' & ');
            document.getElementById('closedMessage').textContent = t('closed_msg', {
                days: dayList,
                start: formatTime(closedMsg.start),
                end: formatTime(closedMsg.end)
            });
        }

        // Sets the phone screen's mode-dependent texts (also re-run on language
        // switch, without clearing what the member already typed).
        function renderIdentifyTexts() {
            document.getElementById('phoneHeading').textContent = mode === 'notify'
                ? t('notify_heading')
                : t('mark_yourself');
            document.getElementById('phoneSub').textContent = mode === 'notify'
                ? t('enter_phone_notify')
                : t('enter_phone');
        }

        function startIdentify(newMode, reason) {
            mode = newMode;
            notifyReason = reason || '';
            selectedMember = null;

            renderIdentifyTexts();
            document.getElementById('phoneInput').value = '';
            document.getElementById('phoneError').textContent = '';
            show('phoneState');
            document.getElementById('phoneInput').focus();
        }

        async function handlePhoneContinue() {
            const phoneError = document.getElementById('phoneError');
            phoneError.textContent = '';

            const raw = document.getElementById('phoneInput').value;
            const digits = raw.replace(/\D/g, '');
            if (digits.length < 9) {
                phoneError.textContent = t('phone_invalid');
                return;
            }

            const btn = document.getElementById('phoneContinueBtn');
            btn.disabled = true;
            btn.textContent = t('searching');

            const { data: result, error } = await supabaseClient
                .rpc('find_members_by_phone', { p_phone: raw });

            btn.disabled = false;
            btn.textContent = t('continue');

            if (error) {
                phoneError.textContent = t('something_wrong') + ' (' + (error.message || error) + ')';
                return;
            }

            if (result?.status !== 'ok' || !Array.isArray(result.members) || result.members.length === 0) {
                phoneError.textContent = t('phone_not_found');
                return;
            }

            enteredPhone = raw;
            if (result.members.length === 1) {
                selectMember(result.members[0]);
            } else {
                renderMemberList(result.members);
                show('pickState');
            }
        }

        function renderMemberList(members) {
            const list = document.getElementById('memberList');
            list.innerHTML = '';
            members.forEach((m) => {
                const row = document.createElement('button');
                row.type = 'button';
                row.className = 'student-row';
                row.innerHTML = `<div class="name">${escapeHtml(maskName(m.name))}</div>`;
                row.addEventListener('click', () => selectMember(m));
                list.appendChild(row);
            });
        }

        function renderConfirmTexts() {
            if (!selectedMember) return;
            document.getElementById('confirmGreeting').textContent = t('greeting', { name: maskName(selectedMember.name) });
            document.getElementById('confirmSub').textContent = mode === 'notify'
                ? t('send_request_q')
                : t('ready_to_mark');
            document.getElementById('confirmBtn').textContent = mode === 'notify'
                ? t('send_to_instructor')
                : t('confirm_checkin');
        }

        function selectMember(member) {
            selectedMember = member;
            renderConfirmTexts();
            document.getElementById('confirmError').textContent = '';
            show('confirmState');
        }

        // The phone number is re-verified INSIDE the database (security definer
        // RPCs): this page never sees anyone's stored phone number, and the
        // check-in RPC also re-enforces the check-in window and records today's
        // date in the studio's timezone (Europe/Rome), not the browser's UTC date.
        async function handleConfirm() {
            if (!selectedMember) return;
            const confirmError = document.getElementById('confirmError');
            confirmError.textContent = '';

            const btn = document.getElementById('confirmBtn');
            btn.disabled = true;
            btn.textContent = mode === 'notify' ? t('sending') : t('checking_in_btn');

            if (mode === 'notify') {
                await submitCheckinRequest(selectedMember, notifyReason);
                btn.disabled = false;
                return;
            }

            const { data: status, error } = await supabaseClient
                .rpc('checkin_self_by_phone', { p_student_id: selectedMember.id, p_phone: enteredPhone });

            btn.disabled = false;
            btn.textContent = t('confirm_checkin');

            if (error) {
                confirmError.textContent = t('something_wrong') + ' (' + (error.message || error) + ')';
                return;
            }

            if (status === 'checked_in') {
                showResult('✓', t('checked_in_title'), t('see_you_in_class', { name: maskName(selectedMember.name) }));
            } else if (status === 'already') {
                showResult('✓', t('already_title'), t('already_sub', { name: maskName(selectedMember.name) }));
            } else if (status === 'closed') {
                // The server-side window check disagrees with this device's clock.
                init();
            } else {
                confirmError.textContent = t('something_wrong');
            }
        }

        // Requests are always reviewed by a human (the instructor taps "Mark
        // Present" per name), and the RPC collapses duplicate same-day requests
        // so nobody can spam.
        async function submitCheckinRequest(member, reason) {
            const { data: status, error } = await supabaseClient
                .rpc('request_manual_checkin_by_phone', {
                    p_student_id: member.id,
                    p_phone: enteredPhone,
                    p_reason: reason || 'Could not complete self check-in'
                });

            if (error) {
                showResult('⚠️', t('notify_failed'), t('try_again_or_ask') + ' (' + (error.message || error) + ')');
                return;
            }

            if (status === 'already') {
                showResult('✓', t('already_title'), t('already_marked_sub', { name: maskName(member.name) }));
                return;
            }

            if (status !== 'sent') {
                showResult('⚠️', t('notify_failed'), t('try_again_or_ask'));
                return;
            }

            showResult('📨', t('instructor_notified'), t('will_mark_manually', { name: maskName(member.name) }));
        }

        function showResult(icon, title, sub) {
            document.getElementById('resultIcon').textContent = icon;
            document.getElementById('resultTitle').textContent = title;
            document.getElementById('resultSub').textContent = sub;
            show('resultState');
        }

        document.addEventListener('DOMContentLoaded', () => {
            document.getElementById('phoneContinueBtn').addEventListener('click', handlePhoneContinue);
            document.getElementById('phoneInput').addEventListener('keydown', (e) => {
                if (e.key === 'Enter') handlePhoneContinue();
            });
            document.getElementById('confirmBtn').addEventListener('click', handleConfirm);
            document.getElementById('confirmBackBtn').addEventListener('click', () => startIdentify(mode, notifyReason));
            document.getElementById('pickBackBtn').addEventListener('click', () => startIdentify(mode, notifyReason));
            document.getElementById('notifyFromClosedBtn').addEventListener('click', () =>
                startIdentify('notify', 'Check-in window closed on their device')
            );
            document.getElementById('notifyFromTooFarBtn').addEventListener('click', () =>
                startIdentify('notify', locBlockStatus === 'unknown'
                    ? 'Device did not share its location'
                    : 'Device location was far from the studio')
            );
            // "I've enabled location — try again": re-run the full check.
            document.getElementById('locRetryBtn').addEventListener('click', () => {
                show('loadingState');
                init();
            });
            document.getElementById('startOverBtn').addEventListener('click', () => {
                // Re-run the full check (window + location) rather than jumping
                // straight to the phone screen, otherwise someone routed to the
                // notify flow for being too far could tap this and land in
                // self-check-in with zero re-verification.
                selectedMember = null;
                init();
            });
            window.addEventListener('popstate', () => {
                selectedMember = null;
                init();
            });

            // Language switch: statics are re-applied by i18n.js; refresh the
            // dynamic (state-dependent) texts of whatever screen is showing,
            // without resetting the flow or clearing the typed number.
            window.onLangChange = () => {
                document.getElementById('branchLine').textContent = t(branchKey);
                renderIdentifyTexts();
                renderConfirmTexts();
                renderClosedMessage();
                renderLocBlock();
            };

            init();
        });
